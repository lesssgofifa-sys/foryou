  <script>
    /* ==========================================================
       PAGE NAVIGATION & INTERACTIVE LOGIC
    ========================================================== */
    let currentStep = 1;
    let sadEmojiInterval = null;

    function updateProgressUI(step) {
      const label = document.getElementById('stepLabel');
      const dot1 = document.getElementById('dot1');
      const dot2 = document.getElementById('dot2');
      const dot3 = document.getElementById('dot3');

      const titles = {
        1: 'A Note From My Heart • 1/3',
        2: 'My Apology • 2/3',
        3: 'With Warm Wishes • 3/3'
      };
      label.textContent = titles[step] || '';

      // Reset dots
      [dot1, dot2, dot3].forEach((dot, idx) => {
        if (idx + 1 === step) {
          dot.className = 'h-2 w-6 rounded-full bg-gradient-to-r from-pink-400 to-sky-400 transition-all duration-300';
        } else if (idx + 1 < step) {
          dot.className = 'h-2 w-2.5 rounded-full bg-pink-400 transition-all duration-300';
        } else {
          dot.className = 'h-2 w-2 rounded-full bg-sky-200 transition-all duration-300';
        }
      });
    }

    function goToPage(targetStep) {
      if (targetStep === currentStep) return;

      const currentEl = document.getElementById(`page${currentStep}`);
      const nextEl = document.getElementById(`page${targetStep}`);

      // Fade out current step
      currentEl.classList.add('fade-out');

      setTimeout(() => {
        currentEl.classList.remove('active', 'fade-out');
        nextEl.classList.add('active');
        currentStep = targetStep;
        updateProgressUI(targetStep);
      }, 280);
    }

    function handleAnswer(choice) {
      const yesBlock = document.getElementById('yesResult');
      const noBlock = document.getElementById('noResult');

      if (sadEmojiInterval) {
        clearInterval(sadEmojiInterval);
        sadEmojiInterval = null;
      }

      if (choice === 'yes') {
        yesBlock.classList.remove('hidden');
        yesBlock.classList.add('flex');
        noBlock.classList.add('hidden');
        noBlock.classList.remove('flex');

        // Trigger vibrant Pink & Blue Confetti immediately
        launchCelebrationConfetti();
      } else {
        noBlock.classList.remove('hidden');
        noBlock.classList.add('flex');
        yesBlock.classList.add('hidden');
        yesBlock.classList.remove('flex');

        // Animate between expressive big sad emojis (🥺 and 😢)
        const sadEl = document.getElementById('bigSadEmoji');
        const sadFaces = ['🥺', '😢', '🥺', '😞'];
        let faceIdx = 0;
        sadEl.textContent = sadFaces[0];
        sadEmojiInterval = setInterval(() => {
          faceIdx = (faceIdx + 1) % sadFaces.length;
          sadEl.textContent = sadFaces[faceIdx];
        }, 2200);
      }

      goToPage(3);
    }

    function resetFlow() {
      if (sadEmojiInterval) {
        clearInterval(sadEmojiInterval);
        sadEmojiInterval = null;
      }
      goToPage(1);
    }

    /* ==========================================================
       CONFETTI ENGINE (PINK, BABY BLUE, LAVENDER & WHITE)
    ========================================================== */
    const confettiCanvas = document.getElementById('confettiCanvas');
    const cCtx = confettiCanvas.getContext('2d');
    let confettiParticles = [];
    let confettiAnimating = false;

    function resizeConfettiCanvas() {
      confettiCanvas.width = window.innerWidth;
      confettiCanvas.height = window.innerHeight;
    }

    const confettiColors = [
      '#ff7eb3', // Pastel Rose Pink
      '#ffafd2', // Soft Baby Pink
      '#7cc6fe', // Baby Blue
      '#a3daff', // Sky Blue
      '#c8b6ff', // Soft Lavender
      '#ffffff', // Pure White
      '#ffe5ec'  // Blush
    ];

    function createConfettiBurst(count = 130) {
      const w = confettiCanvas.width;
      const h = confettiCanvas.height;

      for (let i = 0; i < count; i++) {
        const fromLeft = i % 2 === 0;
        confettiParticles.push({
          x: fromLeft ? w * 0.15 : w * 0.85,
          y: h * 0.55,
          vx: (fromLeft ? 1 : -1) * (Math.random() * 9 + 3) + (Math.random() - 0.5) * 4,
          vy: -(Math.random() * 14 + 7),
          size: Math.random() * 8 + 6,
          color: confettiColors[Math.floor(Math.random() * confettiColors.length)],
          rotation: Math.random() * 360,
          rotationSpeed: (Math.random() - 0.5) * 11,
          gravity: 0.26 + Math.random() * 0.1,
          drag: 0.985,
          wobble: Math.random() * 10,
          wobbleSpeed: 0.08 + Math.random() * 0.06,
          shape: Math.random() > 0.35 ? 'rect' : 'circle',
          opacity: 1
        });
      }
    }

    function animateConfetti() {
      if (confettiParticles.length === 0) {
        confettiAnimating = false;
        cCtx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);
        return;
      }

      confettiAnimating = true;
      cCtx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);

      for (let i = confettiParticles.length - 1; i >= 0; i--) {
        const p = confettiParticles[i];
        p.x += p.vx + Math.sin(p.wobble) * 1.2;
        p.y += p.vy;
        p.vy += p.gravity;
        p.vx *= p.drag;
        p.wobble += p.wobbleSpeed;
        p.rotation += p.rotationSpeed;

        if (p.y > confettiCanvas.height * 0.78) {
          p.opacity -= 0.025;
        }

        cCtx.save();
        cCtx.globalAlpha = Math.max(0, p.opacity);
        cCtx.translate(p.x, p.y);
        cCtx.rotate((p.rotation * Math.PI) / 180);
        cCtx.fillStyle = p.color;

        if (p.shape === 'circle') {
          cCtx.beginPath();
          cCtx.arc(0, 0, p.size * 0.5, 0, Math.PI * 2);
          cCtx.fill();
        } else {
          cCtx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size * 0.65);
        }
        cCtx.restore();

        if (p.y > confettiCanvas.height + 40 || p.opacity <= 0) {
          confettiParticles.splice(i, 1);
        }
      }

      requestAnimationFrame(animateConfetti);
    }

    function launchCelebrationConfetti() {
      resizeConfettiCanvas();
      createConfettiBurst(110);
      setTimeout(() => createConfettiBurst(70), 300);
      if (!confettiAnimating) {
        animateConfetti();
      }
    }

    /* ==========================================================
       AMBIENT FLOATING PASTEL HEARTS & BUBBLES CANVAS
    ========================================================== */
    const bgCanvas = document.getElementById('bgCanvas');
    const bgCtx = bgCanvas.getContext('2d');
    let bgItems = [];

    function initBgParticles() {
      bgCanvas.width = window.innerWidth;
      bgCanvas.height = window.innerHeight;
      bgItems = [];

      const count = window.innerWidth < 640 ? 18 : 28;
      for (let i = 0; i < count; i++) {
        bgItems.push({
          x: Math.random() * bgCanvas.width,
          y: Math.random() * bgCanvas.height,
          size: Math.random() * 10 + 6,
          speedY: -(Math.random() * 0.45 + 0.2),
          speedX: (Math.random() - 0.5) * 0.35,
          alpha: Math.random() * 0.35 + 0.15,
          isHeart: i % 2 === 0,
          color: i % 3 === 0 ? '#ff85b3' : '#7cc6fe'
        });
      }
    }

    function drawMiniHeart(ctx, x, y, size, color, alpha) {
      ctx.save();
      ctx.globalAlpha = alpha;
      ctx.fillStyle = color;
      ctx.beginPath();
      const topCurveHeight = size * 0.3;
      ctx.moveTo(x, y + topCurveHeight);
      ctx.bezierCurveTo(x, y, x - size / 2, y, x - size / 2, y + topCurveHeight);
      ctx.bezierCurveTo(x - size / 2, y + (size + topCurveHeight) / 2, x, y + (size + topCurveHeight) / 1.4, x, y + size);
      ctx.bezierCurveTo(x, y + (size + topCurveHeight) / 1.4, x + size / 2, y + (size + topCurveHeight) / 2, x + size / 2, y + topCurveHeight);
      ctx.bezierCurveTo(x + size / 2, y, x, y, x, y + topCurveHeight);
      ctx.closePath();
      ctx.fill();
      ctx.restore();
    }

    function animateBg() {
      bgCtx.clearRect(0, 0, bgCanvas.width, bgCanvas.height);

      for (const item of bgItems) {
        item.y += item.speedY;
        item.x += item.speedX;

        if (item.y < -25) {
          item.y = bgCanvas.height + 20;
          item.x = Math.random() * bgCanvas.width;
        }
        if (item.x < -25) item.x = bgCanvas.width + 20;
        if (item.x > bgCanvas.width + 25) item.x = -20;

        if (item.isHeart) {
          drawMiniHeart(bgCtx, item.x, item.y, item.size, item.color, item.alpha);
        } else {
          bgCtx.save();
          bgCtx.globalAlpha = item.alpha;
          bgCtx.fillStyle = item.color;
          bgCtx.beginPath();
          bgCtx.arc(item.x, item.y, item.size * 0.38, 0, Math.PI * 2);
          bgCtx.fill();
          bgCtx.restore();
        }
      }

      requestAnimationFrame(animateBg);
    }

    window.addEventListener('resize', () => {
      resizeConfettiCanvas();
      initBgParticles();
    });

    window.onload = function () {
      resizeConfettiCanvas();
      initBgParticles();
      animateBg();
    };
